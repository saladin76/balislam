/** الظرف الصغير بختم القناة الأحمر */
export const Envelope: React.FC<{ x: number; y: number; w?: number; open?: number; coins?: number; rot?: number }> = ({ x, y, w = 120, open = 0, coins = 0, rot = 0 }) => {
  const h = w * 0.62;
  return (
    <g transform={`translate(${x} ${y}) rotate(${rot})`}>
      {coins > 0 ? <g transform={`translate(0 ${-h * 0.5 - 22 * (1 - coins)})`}><ThreeCoins x={0} y={0} r={w * 0.12} tilt={0.55} /></g> : null}
      <rect x={-w / 2} y={-h / 2} width={w} height={h} rx={4} fill={mix(ID.paper, ID.copper, 0.18)} stroke={INK} strokeWidth={3} />
      <path d={`M${-w / 2} ${-h / 2} L0 ${h * 0.08} L${w / 2} ${-h / 2}`} fill="none" stroke={darken(ID.copper, 0.3)} strokeWidth={2.4} />
      {open > 0 ? (
        <path d={`M${-w / 2} ${-h / 2} L0 ${-h / 2 - h * 0.55 * open} L${w / 2} ${-h / 2} Z`} fill={mix(ID.paper, ID.copper, 0.28)} stroke={INK} strokeWidth={3} />

        <circle cx={0} cy={h * 0.05} r={w * 0.1} fill={RED} stroke={darken(RED, 0.4)} strokeWidth={2} />
      )}
    </g>
  );
};

/** ميزان بائع الفاكهة: عمود وعارضة وكفتان ومؤشر؛ tilt بالدرجات (موجب = كفة الفاكهة أثقل) */
export const VendorScale: React.FC<{ x: number; base: number; tilt?: number; thumb?: number; weight?: number; s?: number }> = ({ x, base, tilt = 0, thumb = 0, weight = 0, s = 1 }) => {
  const arm = 120;
  const rad = (tilt * Math.PI) / 180;
  const top = -250;
  const pan = (side: -1 | 1, children: React.ReactNode) => {
    const px = side * arm * Math.cos(rad);
    const py = top + 12 + side * arm * Math.sin(rad);
    return (
      <g transform={`translate(${px} ${py})`}>
        <path d="M0 0 L-40 92 M0 0 L40 92" stroke={darken(ID.copper, 0.2)} strokeWidth={2.4} />
        <path d="M-52 92 Q0 124 52 92 Z" fill={ID.copper} stroke={darken(ID.copper, 0.45)} strokeWidth={3} />
        <g transform="translate(0 92)">{children}</g>
      </g>
    );
  };
  return (
    <g transform={`translate(${x} ${base}) scale(${s})`}>
      <rect x={-8} y={top} width={16} height={-top} fill={darken(ID.copper, 0.15)} stroke={INK} strokeWidth={2} />
      <rect x={-40} y={-14} width={80} height={14} rx={4} fill={darken(ID.copper, 0.3)} stroke={INK} strokeWidth={2} />
      <g transform={`rotate(${tilt} 0 ${top + 12})`}>
        <rect x={-arm - 8} y={top + 6} width={(arm + 8) * 2} height={12} rx={6} fill={ID.copper} stroke={INK} strokeWidth={2} />
      </g>
      <path d={octagramPath(0, top + 12, 13)} fill={ID.copper} stroke={INK} strokeWidth={2} />
      {pan(1, (
        <g>
          {[[-22, -10], [0, -14], [22, -10], [-11, -30], [11, -30]].map(([fx, fy], i) => (
            <circle key={i} cx={fx} cy={fy} r={12} fill={i % 2 ? mix(GOLD, RED, 0.3) : mix(RED, ID.copper, 0.25)} stroke={INK} strokeWidth={2} />
          ))}
          {thumb > 0 ? (
            // إبهام البائع على حافة الكفة: ما رآه الأول
            <g opacity={thumb} transform="scale(1.8) translate(-14 -2)"><path d="M44 -6 q22 -10 40 -4 q6 10 -6 14 q-16 2 -34 -2 Z" fill={mix(ID.copper, ID.paper, 0.45)} stroke={INK} strokeWidth={2.4} /></g>
          ) : null}
        </g>
      ))}
      {pan(-1, weight > 0 ? (
        // العيار المختوم: الدليل
        <g opacity={weight}>
          <path d="M-22 -2 L-16 -38 L16 -38 L22 -2 Z" fill={darken(ID.copper, 0.25)} stroke={INK} strokeWidth={2.4} />
          <circle cx={0} cy={-44} r={9} fill={darken(ID.copper, 0.25)} stroke={INK} strokeWidth={2} />
          <circle cx={0} cy={-20} r={8} fill={RED} />
        </g>
      ) : <path d="M-14 -2 L-10 -26 L10 -26 L14 -2 Z" fill={darken(ID.copper, 0.25)} stroke={INK} strokeWidth={2.4} />)}
    </g>
  );
};

/** فقاعة ما رآه كل شاهد: دائرة مقرّبة بإطار ملوّن */
const Bubble: React.FC<{ cx: number; cy: number; head: [number, number]; ring: string; children: React.ReactNode; r?: number; dim?: number }> = ({ cx, cy, head, ring, children, r = 118, dim = 0 }) => {
  const id = `b${Math.round(cx)}${Math.round(cy)}`;
  const ex = cx + (head[0] < cx ? -r * 0.7 : r * 0.7);
  const ey = cy + r * 0.62;
  return (
    <g opacity={1 - 0.6 * dim}>
      {[0.3, 0.6].map((k, i) => <circle key={i} cx={head[0] + (ex - head[0]) * k} cy={head[1] + (ey - head[1]) * k} r={6 + 6 * k} fill={ID.paper} stroke={ring} strokeWidth={3} />)}
      <defs><clipPath id={id}><circle cx={cx} cy={cy} r={r} /></clipPath></defs>
      <circle cx={cx} cy={cy} r={r + 7} fill={darken(ID.night, 0.2)} opacity={0.35} />
      <g clipPath={`url(#${id})`}><rect x={cx - r} y={cy - r} width={2 * r} height={2 * r} fill={mix(ID.paper, ID.copper, 0.22)} />{children}</g>
      <circle cx={cx} cy={cy} r={r} fill="none" stroke={ring} strokeWidth={7} />
    </g>
  );
};
